python3 << 'EOF'
import os, re, sys
from datetime import datetime
block = '''    location /uploads/ {
        proxy_pass http://127.0.0.1:3000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

'''
for d in ["/www/server/panel/vhost/nginx", "/etc/nginx/sites-enabled", "/etc/nginx/conf.d"]:
    if os.path.isdir(d):
        for f in os.listdir(d):
            p = os.path.join(d, f)
            if os.path.isfile(p) and (f.endswith(".conf") or "120.26.124.180" in f):
                try:
                    c = open(p).read()
                    if "120.26.124.180" in c:
                        if "location /uploads/" in c:
                            print("已存在 /uploads/，跳过"); sys.exit(0)
                        new = re.sub(r'(\s+)(location\s+/\s*\{)', block + r'\1\2', c, 1)
                        if new == c:
                            new = re.sub(r'(server\s*\{)', r'\1\n' + block, c, 1)
                        if new != c:
                            open(p + ".bak", "w").write(c)
                            open(p, "w").write(new)
                            print("已修改:", p)
                            os.system("nginx -t && nginx -s reload")
                            print("完成"); sys.exit(0)
                except Exception as e:
                    pass
print("未找到配置。列出常见路径:")
for d in ["/www/server/panel/vhost/nginx", "/etc/nginx/sites-enabled"]:
    if os.path.isdir(d):
        print(" ", d, ":", os.listdir(d))
sys.exit(1)
EOF
